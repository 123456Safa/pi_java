━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

❌ "Foreign key constraint fails"
   → Vous n'avez pas exécuté le script SQL
   → Allez à ÉTAPE 1 et refaites-la

❌ "Table user doesn't exist"
   → Même problème
   → Allez à ÉTAPE 1 et refaites-la

❌ "Cannot resolve symbol" ou erreur de compilation
   → Allez à ÉTAPE 2 et refaites-la

❌ "Cannot find file"
   → Redémarrez l'application
   → Allez à ÉTAPE 3 et refaites-la

❌ "Impossible de contacter Stripe"
   → Vérifiez votre connexion Internet
   → Vérifiez que api_config.properties a une clé valide
